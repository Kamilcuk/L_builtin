cmake_minimum_required(VERSION 3.14)
project(L_builtin_dispatcher)

# This CMakeLists.txt builds the L_builtin dispatcher from pre-built .so files.

option(L_DISPATCHER_SO_FILES "" "List of pre-built .so files for the dispatcher, e.g., /path/to/L_builtin-5.1.so;/path/to/L_builtin-5.2.so")
if(NOT L_DISPATCHER_SO_FILES)
    message(FATAL_ERROR "L_DISPATCHER_SO_FILES must be set to a list of .so files")
endif()

# We'll process the list of .so files
string(REPLACE ";" "\n" L_DISPATCHER_SO_FILES_LIST "${L_DISPATCHER_SO_FILES}")
set(DISPATCHER_OBJS)
set(VERSION_TABLE_LINES)

foreach(VER_SO IN LISTS L_DISPATCHER_SO_FILES_LIST)
    # Get the filename
    get_filename_component(VER_SO_NAME "${VER_SO}" NAME)
    # We expect the filename to be L_builtin-<version>.so
    if(VER_SO_NAME MATCHES "^L_builtin-(.+)\\.so$")
        set(VER_STRING "${CMAKE_MATCH_1}")
    else()
        message(FATAL_ERROR "Unexpected .so file name: ${VER_SO_NAME}. Expected format: L_builtin-<version>.so")
    endif()

    # Convert version string to a form suitable for symbol names: replace . and - with _
    string(REPLACE "-" "_" VER_STRING_UNDERSCORE ${VER_STRING})
    string(REPLACE "." "_" VER_STRING_UNDERSCORE ${VER_STRING_UNDERSCORE})
    set(OBJCOPY_INPUT_BASE "L_builtin_${VER_STRING_UNDERSCORE}_so")
    set(OBJCOPY_INPUT_NAME "${OBJCOPY_INPUT_BASE}")   # file without extension
    set(OBJCOPY_OUTPUT_NAME "${OBJCOPY_INPUT_BASE}.o")   # L_builtin_5_1_so.o

    set(VER_SO_LOCAL_NOEXT "${CMAKE_CURRENT_BINARY_DIR}/${OBJCOPY_INPUT_NAME}")
    add_custom_command(
        OUTPUT ${VER_SO_LOCAL_NOEXT}
        COMMAND ${CMAKE_COMMAND} -E copy ${VER_SO} ${VER_SO_LOCAL_NOEXT}
        COMMENT "Copying ${VER_SO} to ${VER_SO_LOCAL_NOEXT} for objcopy"
    )

    set(VER_OBJ_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${OBJCOPY_OUTPUT_NAME}")
    add_custom_command(
        OUTPUT ${VER_OBJ_OUTPUT}
        COMMAND ${CMAKE_OBJCOPY} -I binary -O elf64-x86-64 -B i386:x86-64
                --prefix-symbols=_binary_
                ${VER_SO_LOCAL_NOEXT}
                ${VER_OBJ_OUTPUT}
        DEPENDS ${VER_SO_LOCAL_NOEXT}
        COMMENT "Generating object file from ${VER_SO_LOCAL_NOEXT}"
    )

    list(APPEND DISPATCHER_OBJS ${VER_OBJ_OUTPUT})

    # Generate a line for the version table
    list(APPEND VERSION_TABLE_LINES "    {\"${VER_STRING}\", _binary_L_builtin_${VER_STRING_UNDERSCORE}_so_start, _binary_L_builtin_${VER_STRING_UNDERSCORE}_so_end},")
endforeach()

# Generate the version table header file
set(DISPATCHER_VERSION_TABLE_H "${CMAKE_CURRENT_BINARY_DIR}/dispatcher_version_table.h")
file(WRITE "${DISPATCHER_VERSION_TABLE_H}" "/* Generated by CMake - do not edit */\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "#include <stddef.h>\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "struct embedded_so {\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "    const char* version;\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "    const unsigned char* start;\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "    const unsigned char* end;\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "};\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "static struct embedded_so embedded_sos[] = {\n")
foreach(line IN LISTS VERSION_TABLE_LINES)
    file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "${line}\n")
endforeach()
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "    {NULL, NULL, NULL}\n")
file(APPEND "${DISPATCHER_VERSION_TABLE_H}" "};\n")

# Now, create the dispatcher module target
add_library(L_builtin_dispatcher MODULE
    dispatcher.c
    ${DISPATCHER_OBJS}
)
set_target_properties(L_builtin_dispatcher PROPERTIES
    PREFIX "" SUFFIX ".so" OUTPUT_NAME "L_builtin"
)
target_compile_options(L_builtin_dispatcher PRIVATE -ffunction-sections -fdata-sections)
target_link_options(L_builtin_dispatcher PRIVATE -Wl,--gc-sections)
find_package(Threads REQUIRED)
target_link_libraries(L_builtin_dispatcher PRIVATE
    bash ${CMAKE_DL_LIBS} m rt boost_preprocessor bash_loadables
)

# Create a symlink for the main .so target
add_custom_command(TARGET L_builtin_dispatcher POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E create_symlink
        $<TARGET_FILE:L_builtin_dispatcher>
        ${CMAKE_CURRENT_SOURCE_DIR}/L_builtin.so
    COMMENT "Creating symlink to L_builtin.so"
)
